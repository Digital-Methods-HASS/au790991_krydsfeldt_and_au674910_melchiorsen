  #Remove digits

#Code for figure 1
UN_speeches_clean_tfidf |> 
#we uses the clean_tfidf object and pipes anti_join, witch sorts out the stopwords
  anti_join(country_stopwords) |> 
#arrange the values decending
  arrange(desc(tf_idf)) |> 
#modifying the word-column with mutate() to the plot function, with the most common words first by the rev() function.
  mutate(word = factor(word, levels = rev(unique(word)))) |> 
#group by country
  group_by(country) |> 
#selecting the eight most common words after the filtering for each country
  top_n(8) |>  
#plotting and labeling
  ggplot(aes(label = word, size = tf_idf, color = tf_idf)) +
#plotting a wordcloud
  geom_text_wordcloud_area() +
#scaling the size of the words
  scale_size_area(max_size = 13) +
  theme_minimal() +
#ordering the look of the nine wordclouds
  facet_wrap(~country, ncol = 3, scales = "free") +
  scale_color_gradient(low = "darkblue", high = "blue") +
#giving names so the figure is easy to understand.
  labs(
      title = "UN Speeches: most important words pr. country in the timespan from year 1989-2000",
       subtitle = "Importance determined by term frequency (tf) - inversed document frequency(idf)",
